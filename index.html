import os import re import time import requests from bs4 import BeautifulSoup from urllib.parse import urljoin # ===================================================================== # CONFIGURACIÓN INICIAL # Pega aquí la URL de la PÁGINA 1 o el capítulo inicial del libro # ===================================================================== URL_INICIAL = "http://escritos.cavabianca.org/books/meditaciones-iv/chapter/meditaciones-iv" CARPETA_DESTINO = "Obras_Obsidian" # ===================================================================== def limpiar_nombre_archivo(texto): """Elimina caracteres no permitidos en nombres de archivos de Windows/Linux.""" return re.sub(r'[\\/*?:"<>|]', "", texto).strip() def extraer_libro_y_punto(url): """Deduce el nombre del libro y el número de página/punto desde la URL.""" partes = url.split('/') try: idx_books = partes.index('books') nombre_libro = partes[idx_books + 1].capitalize() except ValueError: nombre_libro = "Obra_Desconocida" if 'page' in partes: punto = partes[-1] else: punto = partes[-1] return nombre_libro, punto def buscar_boton_siguiente(soup): """Buscador ultra-robusto bilingüe para la navegación de BookStack.""" # 1. Intentar por atributo técnico estándar rel="next" boton = soup.find('a', attrs={'rel': 'next'}) if boton and boton.has_attr('href'): return boton # 2. Intentar por clases nativas de BookStack para el botón derecho for clase in ['entity-navigation-next', 'navigation-next', 'next', 'text-right']: boton = soup.find('a', class_=re.compile(rf'{clase}', re.IGNORECASE)) if boton and boton.has_attr('href'): return boton # 3. Buscar en todos los enlaces que contengan palabras clave de avance palabras_clave = ['siguiente', 'next', 'sig.', '→', '>]'] for a in soup.find_all('a', href=True): texto_enlace = a.get_text().lower() if any(pc in texto_enlace for pc in palabras_clave): # CORRECCIÓN: Método tradicional .get() compatible con cualquier versión de BS4 clases_ignorar = ['menu', 'profile', 'dropdown', 'header'] clases_enlace = a.get('class', []) # Convertimos a cadena de texto de forma segura clases_str = " ".join(clases_enlace).lower() if isinstance(clases_enlace, list) else str(clases_enlace).lower() if not any(ci in clases_str for ci in clases_ignorar): return a # 4. Búsqueda profunda en contenedores padres si el texto está muy oculto for palabra in ['siguiente', 'next']: elementos = soup.find_all(lambda tag: tag.get_text() and palabra in tag.get_text().lower() and tag.name not in ['script', 'style']) for elem in elementos: if elem.name == 'a' and elem.has_attr('href'): return elem parent_a = elem.find_parent('a') if parent_a and parent_a.has_attr('href'): return parent_a return None def procesar_web(): url_actual = URL_INICIAL capitulo_actual = "Sin Capítulo" session = requests.Session() session.headers.update({ 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36', 'Accept-Language': 'es-ES,es;q=0.9,en;q=0.8' }) print(f"Iniciando descarga en la carpeta: {CARPETA_DESTINO}") while url_actual: print(f"\nProcesando: {url_actual}") try: response = session.get(url_actual, timeout=10) if response.status_code != 200: print(f"Error {response.status_code} al acceder a la página. Fin del proceso.") break except Exception as e: print(f"Error de conexión: {e}") break soup = BeautifulSoup(response.text, 'html.parser') nombre_libro, identificador_punto = extraer_libro_y_punto(url_actual) # PROCESAR SI ES CAMBIO DE CAPÍTULO header_capitulo = soup.find('h1') if header_capitulo and 'chapter' in url_actual: capitulo_actual = header_capitulo.get_text().strip() print(f"--- Nuevo Capítulo Detectado: {capitulo_actual} ---") boton_siguiente = buscar_boton_siguiente(soup) if not boton_siguiente: boton_siguiente = soup.find('a', href=re.compile(r'/page/')) if boton_siguiente and boton_siguiente.has_attr('href'): url_actual = urljoin(url_actual, boton_siguiente['href']) continue else: print("No se encontró forma de avanzar desde este capítulo. Fin.") break # PROCESAR SI ES PÁGINA DE TEXTO (PUNTO) div_contenido = soup.find('div', class_='page-content') or soup.find('div', id='page-content') or soup.find('article') if not div_contenido: print("No se encontró el bloque de contenido principal en esta página.") texto_punto = "Contenido no extraído correctamente. Revisar selectores HTML." else: texto_punto = div_contenido.get_text().strip() ruta_carpeta = os.path.join(CARPETA_DESTINO, limpiar_nombre_archivo(nombre_libro), limpiar_nombre_archivo(capitulo_actual)) os.makedirs(ruta_carpeta, exist_ok=True) nombre_archivo = f"{nombre_libro}_{identificador_punto}.md" ruta_archivo = os.path.join(ruta_carpeta, nombre_archivo) yaml_frontmatter = f"""--- autor: San Josemaría Escrivá obra: {nombre_libro} capitulo: "{capitulo_actual}" punto: {identificador_punto} temas_principales: [] url_origen: {url_actual} --- # {nombre_libro} - Punto {identificador_punto} {texto_punto} """ with open(ruta_archivo, 'w', encoding='utf-8') as f: f.write(yaml_frontmatter) print(f"Guardado con éxito: {nombre_archivo}") # Buscar el botón para la siguiente página boton_siguiente = buscar_boton_siguiente(soup) if boton_siguiente and boton_siguiente.has_attr('href'): url_siguiente = urljoin(url_actual, boton_siguiente['href']) if url_siguiente == url_actual: print("El botón siguiente apunta a la misma página. Finalizando.") url_actual = None else: url_actual = url_siguiente time.sleep(0.4) else: print("No se encontró el botón de avance. El libro ha terminado o requiere revisión.") url_actual = None if __name__ == "__main__": procesar_web()
